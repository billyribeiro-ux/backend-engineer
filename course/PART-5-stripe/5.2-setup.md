## The Stripe Service Module

Already shown in Lesson 5.1. The full version:

```javascript
// services/stripe.js
import Stripe from 'stripe'
import { config } from '../config.js'

export const stripe = new Stripe(config.stripe.secretKey, {
  apiVersion: '2024-04-10',
})

export async function createCustomerForUser(user) {
  const customer = await stripe.customers.create({
    email: user.email,
    name: user.display_name ?? undefined,
    metadata: { user_id: user.id },
  })
  
  // Immediately store in DB
  await db.query(`
    UPDATE users SET stripe_customer_id = $1 WHERE id = $2
  `, [customer.id, user.id])
  
  return customer
}

export async function syncProductToStripe(product) {
  const stripeProduct = await stripe.products.create({
    name: product.name,
    description: product.description ?? undefined,
    metadata: { product_id: product.id },
  })
  
  await db.query(`
    UPDATE products SET stripe_product_id = $1 WHERE id = $2
  `, [stripeProduct.id, product.id])
  
  return stripeProduct
}

export async function syncPriceToStripe(price, stripeProductId, isRecurring = false, interval, intervalCount) {
  const priceData = {
    product: stripeProductId,
    unit_amount: price.amount_cents,
    currency: price.currency,
  }
  
  if (isRecurring) {
    priceData.recurring = { interval, interval_count: intervalCount }
  }
  
  const stripePrice = await stripe.prices.create(priceData)
  
  await db.query(`
    UPDATE prices SET stripe_price_id = $1 WHERE id = $2
  `, [stripePrice.id, price.id])
  
  return stripePrice
}
```

## Syncing Existing Products on Startup

When you first set up Stripe, sync any existing products and plans:

```javascript
// scripts/sync-stripe.js (run once)
async function syncAll() {
  const products = await db.query(`SELECT * FROM products WHERE stripe_product_id IS NULL`)
  
  for (const product of products.rows) {
    const stripeProduct = await syncProductToStripe(product)
    
    const prices = await db.query(`
      SELECT * FROM prices WHERE product_id = $1 AND stripe_price_id IS NULL
    `, [product.id])
    
    for (const price of prices.rows) {
      await syncPriceToStripe(price, stripeProduct.id)
    }
  }
  
  // Same for plans...
}
```

This lesson is short — the setup is straightforward. The complexity is in the webhook handling (Lesson 5.5) and the subscription flow (Lesson 5.4).
