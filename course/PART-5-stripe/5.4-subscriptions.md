# Lesson 5.4 — Subscriptions

## Creating a Subscription

The full subscription creation flow was shown in Lesson 4.3. The Stripe-specific parts:

```javascript
// The key Stripe API calls

// 1. Attach payment method to customer
await stripe.paymentMethods.attach(payment_method_id, {
  customer: stripe_customer_id,
})

// 2. Set as default payment method
await stripe.customers.update(stripe_customer_id, {
  invoice_settings: { default_payment_method: payment_method_id },
})

// 3. Create the subscription
const stripeSubscription = await stripe.subscriptions.create({
  customer: stripe_customer_id,
  items: [{ price: stripe_price_id }],
  trial_period_days: 14,                              // free trial
  coupon: stripe_coupon_id ?? undefined,              // discount
  payment_behavior: 'default_incomplete',             // don't charge yet if payment might fail
  payment_settings: {
    save_default_payment_method: 'on_subscription',
  },
  expand: ['latest_invoice.payment_intent'],          // get PaymentIntent in response
})
```

## Cancellation

```javascript
export async function cancelSubscription(req, res) {
  const { immediately = false } = req.body
  const subscription = req.resources[0]  // fetched by requirePolicy middleware
  
  let stripeSubscription
  
  if (immediately) {
    // Cancel immediately — user loses access now
    stripeSubscription = await stripe.subscriptions.cancel(
      subscription.stripe_subscription_id
    )
  } else {
    // Cancel at period end — user keeps access until billing period ends
    stripeSubscription = await stripe.subscriptions.update(
      subscription.stripe_subscription_id,
      { cancel_at_period_end: true }
    )
  }
  
  // Update local record
  await db.query(`
    UPDATE subscriptions
    SET
      status = $1,
      cancel_at_period_end = $2,
      canceled_at = CASE WHEN $3 THEN NOW() ELSE canceled_at END,
      updated_at = NOW()
    WHERE id = $4
  `, [
    stripeSubscription.status,
    stripeSubscription.cancel_at_period_end,
    immediately,
    subscription.id,
  ])
  
  // Audit log
  await db.auditLogs.create({
    actor_id: req.user.id,
    action: 'subscription.canceled',
    resource_type: 'subscription',
    resource_id: subscription.id,
    metadata: { immediately, canceled_by: req.user.id }
  })
  
  return res.json({
    subscription: await db.subscriptions.findById(subscription.id)
  })
}
```

## Resuming a Canceled Subscription

If a user set `cancel_at_period_end = true` but changes their mind before the period ends:

```javascript
export async function resumeSubscription(req, res) {
  const subscription = req.resources[0]
  
  if (!subscription.cancel_at_period_end) {
    return res.status(409).json({
      error: { code: 'NOT_CANCELING', message: 'This subscription is not set to cancel.' }
    })
  }
  
  const stripeSubscription = await stripe.subscriptions.update(
    subscription.stripe_subscription_id,
    { cancel_at_period_end: false }
  )
  
  await db.query(`
    UPDATE subscriptions SET cancel_at_period_end = false, updated_at = NOW() WHERE id = $1
  `, [subscription.id])
  
  return res.json({ subscription: await db.subscriptions.findById(subscription.id) })
}
```

## Subscription Status and Access

```javascript
// utils/access.js

// Statuses that mean "has access"
const ACCESS_STATUSES = new Set(['active', 'trialing'])

// Statuses that have a grace period (past_due: Stripe retries for several days)
const GRACE_STATUSES = new Set(['past_due'])

export function hasSubscriptionAccess(subscription) {
  if (!subscription) return false
  if (ACCESS_STATUSES.has(subscription.status)) return true
  
  // Grace period for past_due — check if grace period has expired
  if (GRACE_STATUSES.has(subscription.status)) {
    const graceDaysMs = 3 * 24 * 60 * 60 * 1000  // 3 days grace
    const periodEnd = new Date(subscription.current_period_end)
    return (Date.now() - periodEnd.getTime()) < graceDaysMs
  }
  
  return false
}
```

## The Subscription-Access Decision Table

| Status | Access? | Notes |
|--------|---------|-------|
| `trialing` | Yes | Trial period active |
| `active` | Yes | Paying and current |
| `past_due` | Yes (grace) | Payment failed, Stripe retrying |
| `unpaid` | No | Retries exhausted |
| `canceled` | No | Deliberately canceled (or `cancel_at_period_end` period ended) |
| `incomplete` | No | Initial payment pending |
| `incomplete_expired` | No | Initial payment never completed |

The webhook handles all status transitions. Your code never manually decides "this subscription should be active" — Stripe tells you, via webhook.
