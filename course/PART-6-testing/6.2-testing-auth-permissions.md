## Testing RBAC: Role Gates

For every protected endpoint, test that each role gets the correct response:

```javascript
// tests/integration/rbac.test.js
import request from 'supertest'
import app from '../../src/app.js'

// Helper: log in as a test user and get their access token
async function loginAs(email) {
  const res = await request(app)
    .post('/api/v1/auth/login')
    .send({ email, password: 'TestPassword123!' })
  return res.body.access_token
}

describe('RBAC: Product Price Management', () => {
  let adminToken, memberToken, supportToken
  
  beforeAll(async () => {
    adminToken   = await loginAs('admin@test.com')
    memberToken  = await loginAs('member@test.com')
    supportToken = await loginAs('support@test.com')
  })
  
  // POST /products/:id/prices — requires prices:write (admin only)
  describe('POST /products/:id/prices', () => {
    const productId = 'some-existing-product-id'  // from test seed
    
    test('admin can add a price', async () => {
      const res = await request(app)
        .post(`/api/v1/products/${productId}/prices`)
        .set('Authorization', `Bearer ${adminToken}`)
        .send({ amount_cents: 3999, applies_to: 'future_members' })
      
      expect(res.status).toBe(201)
    })
    
    test('member cannot add a price', async () => {
      const res = await request(app)
        .post(`/api/v1/products/${productId}/prices`)
        .set('Authorization', `Bearer ${memberToken}`)
        .send({ amount_cents: 3999, applies_to: 'future_members' })
      
      expect(res.status).toBe(403)
      expect(res.body.error.code).toBe('INSUFFICIENT_PERMISSION')
    })
    
    test('support cannot add a price', async () => {
      const res = await request(app)
        .post(`/api/v1/products/${productId}/prices`)
        .set('Authorization', `Bearer ${supportToken}`)
        .send({ amount_cents: 3999, applies_to: 'future_members' })
      
      expect(res.status).toBe(403)
    })
    
    test('unauthenticated request is rejected', async () => {
      const res = await request(app)
        .post(`/api/v1/products/${productId}/prices`)
        .send({ amount_cents: 3999, applies_to: 'future_members' })
      
      expect(res.status).toBe(401)
    })
  })
})
```

## Testing ABAC: Ownership Checks

```javascript
// tests/integration/abac.test.js

describe('ABAC: Profile Editing', () => {
  let memberToken1, memberToken2, adminToken
  let user1Id, user2Id
  
  beforeAll(async () => {
    memberToken1 = await loginAs('member@test.com')
    memberToken2 = await loginAs('member2@test.com')  // assumes another test member
    adminToken   = await loginAs('admin@test.com')
    
    // Get user IDs
    const me1 = await request(app)
      .get('/api/v1/users/me')
      .set('Authorization', `Bearer ${memberToken1}`)
    user1Id = me1.body.id
    
    const me2 = await request(app)
      .get('/api/v1/users/me')
      .set('Authorization', `Bearer ${memberToken2}`)
    user2Id = me2.body.id
  })
  
  test('member can edit their own profile', async () => {
    const res = await request(app)
      .patch(`/api/v1/users/${user1Id}`)
      .set('Authorization', `Bearer ${memberToken1}`)
      .send({ display_name: 'Updated Name' })
    
    expect(res.status).toBe(200)
    expect(res.body.user.display_name).toBe('Updated Name')
  })
  
  test('member CANNOT edit another member profile', async () => {
    const res = await request(app)
      .patch(`/api/v1/users/${user2Id}`)  // user1's token, user2's ID
      .set('Authorization', `Bearer ${memberToken1}`)
      .send({ display_name: 'Hacked Name' })
    
    expect(res.status).toBe(403)
    
    // Verify the name was NOT changed
    const check = await request(app)
      .get(`/api/v1/users/${user2Id}`)
      .set('Authorization', `Bearer ${adminToken}`)
    expect(check.body.user.display_name).not.toBe('Hacked Name')
  })
  
  test('admin CAN edit any profile', async () => {
    const res = await request(app)
      .patch(`/api/v1/users/${user1Id}`)
      .set('Authorization', `Bearer ${adminToken}`)
      .send({ display_name: 'Admin Updated' })
    
    expect(res.status).toBe(200)
  })
})
```

## Testing Subscription Gates

```javascript
describe('ABAC: Subscription Access Gate', () => {
  let activeToken, trialToken, lapsedToken
  
  beforeAll(async () => {
    activeToken  = await loginAs('member@test.com')          // has active sub (from test seed)
    trialToken   = await loginAs('member-trial@test.com')    // has trialing sub
    lapsedToken  = await loginAs('member-lapsed@test.com')   // has past_due sub (expired grace)
  })
  
  describe('GET /content/premium — requires active subscription', () => {
    test('active subscriber has access', async () => {
      const res = await request(app)
        .get('/api/v1/content/premium')
        .set('Authorization', `Bearer ${activeToken}`)
      
      expect(res.status).toBe(200)
    })
    
    test('trial subscriber has access', async () => {
      const res = await request(app)
        .get('/api/v1/content/premium')
        .set('Authorization', `Bearer ${trialToken}`)
      
      expect(res.status).toBe(200)
    })
    
    test('lapsed subscriber has NO access', async () => {
      const res = await request(app)
        .get('/api/v1/content/premium')
        .set('Authorization', `Bearer ${lapsedToken}`)
      
      expect(res.status).toBe(403)
      expect(res.body.error.code).toBe('SUBSCRIPTION_REQUIRED')
      expect(res.body.error.subscription_status).toBe('past_due')
    })
  })
})
```

## Testing the Auth Flow End-to-End

```javascript
describe('Auth: Full Login/Refresh/Logout cycle', () => {
  test('complete auth flow', async () => {
    // Register
    const register = await request(app)
      .post('/api/v1/auth/register')
      .send({ email: 'flow@test.com', password: 'Pass123!', display_name: 'Flow User' })
    expect(register.status).toBe(201)
    
    const accessToken = register.body.access_token
    const refreshCookie = register.headers['set-cookie']
    
    // Use access token
    const me = await request(app)
      .get('/api/v1/users/me')
      .set('Authorization', `Bearer ${accessToken}`)
    expect(me.status).toBe(200)
    expect(me.body.email).toBe('flow@test.com')
    
    // Refresh tokens
    const refresh = await request(app)
      .post('/api/v1/auth/refresh')
      .set('Cookie', refreshCookie)
    expect(refresh.status).toBe(200)
    expect(refresh.body.access_token).toBeTruthy()
    
    // Old access token still works (it's still valid — 15 min)
    // New access token works
    const meAgain = await request(app)
      .get('/api/v1/users/me')
      .set('Authorization', `Bearer ${refresh.body.access_token}`)
    expect(meAgain.status).toBe(200)
    
    // Logout
    const logout = await request(app)
      .post('/api/v1/auth/logout')
      .set('Authorization', `Bearer ${refresh.body.access_token}`)
      .set('Cookie', refreshCookie)
    expect(logout.status).toBe(204)
    
    // Cleanup
    await db.query(`DELETE FROM users WHERE email = 'flow@test.com'`)
  })
})
```
