# Lesson 7.2 — Adding the Organizations Table

## The Migration Strategy

The `organizations` table already exists (we scaffolded it in Lesson 1.2). What we need to do now is:

1. Add `organization_id` columns to all org-scoped tables
2. Create default organizations for existing users
3. Backfill the `organization_id` values
4. Add NOT NULL constraints after backfilling

This is a **multi-step migration** — adding a NOT NULL column to an existing table requires a safe sequence of changes.

---

## Safe Migration for Adding NOT NULL Columns to Existing Tables

**The dangerous way (never do this in production):**
```sql
ALTER TABLE users ADD COLUMN organization_id UUID NOT NULL REFERENCES organizations(id);
```
This fails if there are existing rows — they have no value for `organization_id`, violating the NOT NULL constraint.

**The safe way (3 steps):**

```sql
-- Step 1: Add the column as nullable
ALTER TABLE users ADD COLUMN organization_id UUID REFERENCES organizations(id);

-- Step 2: Backfill existing rows
-- (insert a default org or assign to an existing org)

-- Step 3: Add NOT NULL constraint (after all rows have values)
ALTER TABLE users ALTER COLUMN organization_id SET NOT NULL;
```

In production, each step is a separate migration run. Steps 1 and 2 can deploy with no downtime. Step 3 requires a brief lock (but it's fast if all rows are filled).

## The Multi-Tenancy Migration

```sql
-- migrations/002_add_multi_tenancy.sql

-- ============================================================
-- Step 1: Add organization_id to org-scoped tables (nullable)
-- ============================================================

ALTER TABLE users ADD COLUMN IF NOT EXISTS organization_id UUID REFERENCES organizations(id);
ALTER TABLE subscriptions ADD COLUMN IF NOT EXISTS organization_id UUID REFERENCES organizations(id);
ALTER TABLE invoices ADD COLUMN IF NOT EXISTS organization_id UUID REFERENCES organizations(id);
ALTER TABLE payments ADD COLUMN IF NOT EXISTS organization_id UUID REFERENCES organizations(id);
ALTER TABLE audit_logs ADD COLUMN IF NOT EXISTS organization_id UUID REFERENCES organizations(id);

-- ============================================================
-- Step 2: Create organization_members table
-- ============================================================

CREATE TABLE IF NOT EXISTS organization_members (
  id              UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  organization_id UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,
  user_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  role_id         UUID NOT NULL REFERENCES roles(id),
  invited_by      UUID REFERENCES users(id),
  joined_at       TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  UNIQUE (organization_id, user_id)
);

CREATE INDEX idx_org_members_org ON organization_members(organization_id);
CREATE INDEX idx_org_members_user ON organization_members(user_id);

-- ============================================================
-- Step 3 (DATA MIGRATION — run in application code or manually)
-- Create a default organization for every existing user
-- and assign them to it.
-- This is done in a seed/migration script, not pure SQL,
-- because it requires generating UUIDs and linking records.
-- ============================================================
```

The data migration is run as a script:

```javascript
// migrations/scripts/backfill-organizations.js
async function backfillOrganizations() {
  const adminRole = await db.query(`SELECT id FROM roles WHERE name = 'admin'`)
  const adminRoleId = adminRole.rows[0].id
  
  // Find all users without an organization
  const users = await db.query(`
    SELECT id, email, display_name FROM users WHERE organization_id IS NULL AND deleted_at IS NULL
  `)
  
  for (const user of users.rows) {
    await db.transaction(async (client) => {
      // Create a default org for this user
      const org = await client.query(`
        INSERT INTO organizations (id, name, slug, is_active)
        VALUES (uuid_generate_v4(), $1, $2, true)
        RETURNING id
      `, [
        `${user.display_name ?? user.email}'s Organization`,
        user.email.replace('@', '-at-').replace(/[^a-z0-9-]/g, '-'),
      ])
      
      const orgId = org.rows[0].id
      
      // Assign user to the org as admin
      await client.query(`
        INSERT INTO organization_members (id, organization_id, user_id, role_id)
        VALUES (uuid_generate_v4(), $1, $2, $3)
        ON CONFLICT DO NOTHING
      `, [orgId, user.id, adminRoleId])
      
      // Set as user's primary org
      await client.query(`
        UPDATE users SET organization_id = $1 WHERE id = $2
      `, [orgId, user.id])
      
      // Update related records
      await client.query(`
        UPDATE subscriptions SET organization_id = $1 WHERE user_id = $2 AND organization_id IS NULL
      `, [orgId, user.id])
      
      await client.query(`
        UPDATE invoices SET organization_id = $1 WHERE user_id = $2 AND organization_id IS NULL
      `, [orgId, user.id])
    })
  }
  
  console.log(`Backfilled ${users.rows.length} users into organizations`)
}
```

After running the backfill:
```sql
-- Step 4: Make organization_id NOT NULL
ALTER TABLE users ALTER COLUMN organization_id SET NOT NULL;
ALTER TABLE subscriptions ALTER COLUMN organization_id SET NOT NULL;
ALTER TABLE invoices ALTER COLUMN organization_id SET NOT NULL;
ALTER TABLE payments ALTER COLUMN organization_id SET NOT NULL;
-- Note: audit_logs stays nullable (system-level logs have no org)
```

## Indexes for Tenant Isolation

Add indexes on `organization_id` for every table that has it. These are the most-used indexes in a multi-tenant system:

```sql
CREATE INDEX idx_users_org ON users(organization_id) WHERE deleted_at IS NULL;
CREATE INDEX idx_subscriptions_org ON subscriptions(organization_id) WHERE deleted_at IS NULL;
CREATE INDEX idx_invoices_org ON invoices(organization_id);
CREATE INDEX idx_payments_org ON payments(organization_id);
CREATE INDEX idx_audit_logs_org ON audit_logs(organization_id);
```

Without these indexes, every tenant-scoped query does a full table scan. With a large table (100k+ rows), this becomes catastrophically slow.

## The Org Middleware

After the migration, add middleware that attaches the current org to every request:

```javascript
// middleware/resolveOrganization.js

export async function resolveOrganization(req, res, next) {
  if (!req.user) return next()  // unauthenticated requests have no org context
  
  // Get org from JWT or header
  let orgId = req.user.organization_id  // from JWT payload
  
  // If user provides X-Organization-Id header (for multi-org users), validate they belong to it
  const requestedOrgId = req.headers['x-organization-id']
  if (requestedOrgId) {
    const membership = await db.query(`
      SELECT 1 FROM organization_members
      WHERE organization_id = $1 AND user_id = $2
      LIMIT 1
    `, [requestedOrgId, req.user.id])
    
    if (!membership.rows.length) {
      return res.status(403).json({
        error: { code: 'ORG_ACCESS_DENIED', message: 'You do not belong to this organization.' }
      })
    }
    
    orgId = requestedOrgId
  }
  
  req.organizationId = orgId
  next()
}
```
