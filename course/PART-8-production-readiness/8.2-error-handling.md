# Lesson 8.2 — Error Handling Strategy

## The Golden Rule

**Never expose internal errors to clients.**

An internal error message like `column "password_hsh" does not exist` reveals:
- Your database schema (attack surface)
- That you have a typo in your code (incompetence signal)
- Potentially the query structure (SQL injection hints)

A client-facing error should be safe, predictable, and useful to the user — not to an attacker.

---

## Error Categories

| HTTP Status | Category | What it means | How client handles it |
|-------------|---------|--------------|----------------------|
| 400 | Bad Request | You sent invalid data | Show validation errors |
| 401 | Unauthorized | Not authenticated | Redirect to login |
| 403 | Forbidden | Authenticated but not allowed | Show access denied |
| 404 | Not Found | Resource doesn't exist | Show not found page |
| 409 | Conflict | Action conflicts with current state | Show specific error |
| 422 | Unprocessable | Data format correct but semantically wrong | Show specific error |
| 429 | Too Many Requests | Rate limited | Show retry-after message |
| 500 | Internal Server Error | Something went wrong on our side | Show generic error, retry |

## The Centralized Error Handler

```javascript
// middleware/errorHandler.js

export class AppError extends Error {
  constructor(statusCode, code, message, field = null) {
    super(message)
    this.statusCode = statusCode
    this.code = code
    this.field = field
  }
}

// Convenience constructors
export const Errors = {
  notFound: (resource = 'Resource') =>
    new AppError(404, 'NOT_FOUND', `${resource} not found.`),
  
  forbidden: (message = 'Access denied.') =>
    new AppError(403, 'FORBIDDEN', message),
  
  unauthorized: (message = 'Authentication required.') =>
    new AppError(401, 'UNAUTHORIZED', message),
  
  conflict: (code, message) =>
    new AppError(409, code, message),
  
  badRequest: (code, message, field = null) =>
    new AppError(400, code, message, field),
  
  internal: () =>
    new AppError(500, 'INTERNAL_ERROR', 'An unexpected error occurred. Please try again.'),
}

// Global error handler middleware (must have 4 arguments to be recognized as error handler)
export function errorHandler(err, req, res, next) {
  // Log the full error internally (with stack trace, request details)
  console.error({
    error: err.message,
    stack: err.stack,
    code: err.code,
    url: req.url,
    method: req.method,
    user_id: req.user?.id,
    organization_id: req.organizationId,
  })
  
  // Known application errors
  if (err instanceof AppError) {
    return res.status(err.statusCode).json({
      error: {
        code: err.code,
        message: err.message,
        field: err.field ?? undefined,
      }
    })
  }
  
  // Stripe errors
  if (err.type === 'StripeCardError') {
    return res.status(400).json({
      error: {
        code: 'PAYMENT_FAILED',
        message: err.message,  // Stripe card errors are safe to show users
      }
    })
  }
  
  if (err.type?.startsWith('Stripe')) {
    // Internal Stripe error — don't expose details
    return res.status(500).json({
      error: {
        code: 'PAYMENT_SERVICE_ERROR',
        message: 'Payment service unavailable. Please try again.',
      }
    })
  }
  
  // PostgreSQL errors
  if (err.code === '23505') {
    // Unique constraint violation — figure out which field
    const field = extractFieldFromUniqueViolation(err)
    return res.status(409).json({
      error: {
        code: 'DUPLICATE_VALUE',
        message: `This ${field ?? 'value'} already exists.`,
        field,
      }
    })
  }
  
  if (err.code === '23503') {
    // Foreign key violation
    return res.status(400).json({
      error: {
        code: 'INVALID_REFERENCE',
        message: 'Referenced resource does not exist.',
      }
    })
  }
  
  // Unknown error — log it and return a safe generic message
  // In production: also send to error tracking (Sentry, Datadog, etc.)
  return res.status(500).json({
    error: {
      code: 'INTERNAL_ERROR',
      message: 'An unexpected error occurred. Please try again later.',
    }
  })
}

function extractFieldFromUniqueViolation(err) {
  // err.detail is something like: 'Key (email)=(test@test.com) already exists.'
  const match = err.detail?.match(/Key \(([^)]+)\)/)
  return match?.[1] ?? null
}
```

Register it at the bottom of your Express app (after all routes):
```javascript
// app.js
app.use('/api/v1', routes)
app.use(errorHandler)  // Must be last
```

## Using the Error Handler in Handlers

```javascript
import { Errors } from '../middleware/errorHandler.js'

export async function getSubscription(req, res, next) {
  try {
    const subscription = await db.subscriptions.findById(req.params.id, req.organizationId)
    
    if (!subscription) {
      // Use the error factory — clean, consistent
      throw Errors.notFound('Subscription')
    }
    
    res.json({ subscription })
    
  } catch (err) {
    next(err)  // Pass to error handler
  }
}
```

Or use an `asyncHandler` wrapper to avoid try/catch everywhere:

```javascript
// utils/asyncHandler.js
export function asyncHandler(fn) {
  return (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next)
  }
}
```

```javascript
// routes/subscriptions.js
import { asyncHandler } from '../utils/asyncHandler.js'

router.get('/subscriptions/:id',
  authenticate,
  asyncHandler(getSubscription)  // No try/catch needed in the handler
)
```

## Validation Errors

Input validation errors (missing fields, invalid formats) should be specific:

```javascript
import Joi from 'joi'  // or zod, or yup

const createSubscriptionSchema = Joi.object({
  plan_id: Joi.string().uuid().required(),
  payment_method_id: Joi.string().required(),
  coupon_code: Joi.string().optional(),
})

export function validate(schema) {
  return (req, res, next) => {
    const { error } = schema.validate(req.body, { abortEarly: false })
    
    if (error) {
      const fields = error.details.map(d => ({
        field: d.path.join('.'),
        message: d.message,
      }))
      
      return res.status(400).json({
        error: {
          code: 'VALIDATION_ERROR',
          message: 'Invalid request data.',
          fields,
        }
      })
    }
    
    next()
  }
}
```

## Error Logging in Production

In production, send errors to an error tracking service. The two most common:

**Sentry:**
```javascript
import * as Sentry from '@sentry/node'

Sentry.init({ dsn: config.sentryDsn })

// In error handler, before the generic response:
if (err.statusCode >= 500 || !(err instanceof AppError)) {
  Sentry.captureException(err, {
    user: { id: req.user?.id, email: req.user?.email },
    extra: { url: req.url, method: req.method },
  })
}
```

**The SENTRY_DSN** goes in `.env.example`:
```bash
SENTRY_DSN=https://your-key@sentry.io/your-project-id
```

Production principle: if a 500 error fires, a developer should know within minutes. Not from a user report — from your monitoring.
